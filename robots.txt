# flarestamina.com

User-agent: *
Allow: /

# Google needs the stylesheet, the script and the fonts to render a page the
# way a student sees it. Blocking them makes every page look broken to the
# crawler, so they are explicitly open.
Allow: /assets/
Allow: /news/news.css

# Private area, passcode-gated and unlinked.
Disallow: /admin/

# Homepage v2 work in progress. The page also carries a noindex meta tag, so
# this line is belt-and-braces: it keeps the crawler from spending budget here.
Disallow: /staging/

# /posts/ carries a noindex instead of a Disallow on purpose: a disallowed URL
# can still be indexed without its content, because the crawler never gets to
# read the noindex.

# A plain-language map of the site for assistants that read one.
# chatgpt.com is the second referrer this week, level with Google.
